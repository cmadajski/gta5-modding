find_program(NATIVE_MOTION_DXC NAMES dxc DOC "Host DXC compiler for the SM6 wheel fixture" NO_CMAKE_FIND_ROOT_PATH)
if(NOT NATIVE_MOTION_DXC)
    message(STATUS "Wheel override GPU fixture needs a host DXC compiler (NATIVE_MOTION_DXC)")
    return()
endif()
foreach(stage IN ITEMS vertex pixel)
    if(stage STREQUAL "vertex")
        set(profile vs_6_0)
    else()
        set(profile ps_6_0)
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${stage}.cso"
        COMMAND "${NATIVE_MOTION_DXC}" -T "${profile}" -E "${stage}"
            "${CMAKE_CURRENT_SOURCE_DIR}/fixture.hlsl" -Fo "${CMAKE_CURRENT_BINARY_DIR}/${stage}.cso"
        DEPENDS fixture.hlsl VERBATIM)
endforeach()
add_custom_target(wheel_override_shaders DEPENDS
    "${CMAKE_CURRENT_BINARY_DIR}/vertex.cso" "${CMAKE_CURRENT_BINARY_DIR}/pixel.cso")
